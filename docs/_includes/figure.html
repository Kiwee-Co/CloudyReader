{%- comment -%} A figure for an image in /assets/img/. Parameters: src (a file name there, or a site path that starts with "/"), alt (required, never empty), caption (optional, Markdown), width and height (optional, in CSS pixels, so the page reserves the space). A "-light" file gets its "-dark" file as the dark-appearance source; any other file gets one when a "-dark" sibling exists. {%- endcomment -%} {%- assign figure_src = include.src | strip -%} {%- unless figure_src contains "/" -%}{%- assign figure_src = "/assets/img/" | append: figure_src -%}{%- endunless -%} {%- assign figure_dark = "" -%} {%- if figure_src contains "-light." -%} {%- assign figure_dark = figure_src | replace: "-light.", "-dark." -%} {%- else -%} {%- assign figure_ext = figure_src | split: "." | last -%} {%- assign figure_stem_size = figure_src.size | minus: figure_ext.size | minus: 1 -%} {%- assign figure_sibling = figure_src | slice: 0, figure_stem_size | append: "-dark." | append: figure_ext -%} {%- assign figure_found = site.static_files | where: "path", figure_sibling | first -%} {%- if figure_found -%}{%- assign figure_dark = figure_sibling -%}{%- endif -%} {%- endif -%}
